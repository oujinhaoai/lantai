func (ev *m211Evidence) save(name string, value any) {
	ev.t.Helper()
	if ev.dir == "" {
		return
	}
	raw, err := json.MarshalIndent(m211Redact(value), "", "  ")
	if err != nil {
		ev.t.Fatal(err)
	}
	if err = os.WriteFile(filepath.Join(ev.dir, name+".json"), append(raw, '\n'), 0600); err != nil {
		ev.t.Fatal(err)
	}
}

// Raw machine fields are retained; transient capability URLs and credentials
// are omitted even though every account and byte in this run is synthetic.
func m211Redact(value any) any {
	raw, _ := json.Marshal(value)
	var out any
	_ = json.Unmarshal(raw, &out)
	var clean func(any)
	clean = func(v any) {
		switch v := v.(type) {
		case map[string]any:
			for k, x := range v {
				switch k {
				case "token", "csrf_token", "password", "secret", "code", "url", "part_url_template", "parts_url":
					if k == "code" {
						clean(x)
					} else {
						v[k] = "[redacted]"
					}
				default:
					clean(x)
				}
			}
		case []any:
			for _, x := range v {
				clean(x)
			}
		}
	}
	clean(out)
	return out
}

type m211StatusWriter struct {
	http.ResponseWriter
	status int
	count  int
}

func (w *m211StatusWriter) WriteHeader(code int) { w.status = code; w.ResponseWriter.WriteHeader(code) }
func (w *m211StatusWriter) Write(b []byte) (int, error) {
	if w.status == 0 {
		w.WriteHeader(200)
	}
	n, err := w.ResponseWriter.Write(b)
	w.count += n
	return n, err
}

func m211HTTP(t *testing.T, h *consistencyClient, method, path, key string, body any, data []byte) ([]byte, int) {
	t.Helper()
	u, err := url.Parse(path)
	if err != nil {
		t.Fatal("invalid protocol path")
	}
	base, _ := url.Parse(h.origin)
	u = base.ResolveReference(u)
	if u.Host != base.Host || u.Scheme != base.Scheme {
		t.Fatal("cross-origin protocol URL")
	}
	var input io.Reader
	if data != nil {
		input = bytes.NewReader(data)
	} else if body != nil {
		input = bytes.NewReader(mustJSON(body))
	}
	req, err := http.NewRequestWithContext(t.Context(), method, u.String(), input)
	if err != nil {
		t.Fatal(err)
	}
	req.Header.Set("Authorization", "Bearer "+h.token)
	if key != "" {
		req.Header.Set("Idempotency-Key", key)
	}
	if data != nil {
		req.Header.Set("Lantai-Part-Sha256", shaOf(data))
	} else if body != nil {
		req.Header.Set("Content-Type", "application/json")
	}
	tr := &http.Transport{}
	defer tr.CloseIdleConnections()
	r, err := (&http.Client{Transport: tr, Timeout: 30 * time.Second, CheckRedirect: func(*http.Request, []*http.Request) error { return http.ErrUseLastResponse }}).Do(req)
	if err != nil {
		t.Fatal("loopback HTTP failed")
	}
	defer r.Body.Close()
	raw, err := io.ReadAll(r.Body)
	if err != nil {
		t.Fatal(err)
	}
	return raw, r.StatusCode
}

func m211Call(t *testing.T, h *consistencyClient, tool string, args map[string]any) (map[string]any, bool) {
	t.Helper()
	out, err := h.mcp.CallTool(t.Context(), &mcp.CallToolParams{Name: tool, Arguments: args})
	if err != nil {
		t.Fatal("MCP protocol failure", err)
	}
	raw := mustJSON(out)
	if bytes.Contains(raw, []byte(h.token)) || bytes.Contains(raw, []byte("/xfer/")) || bytes.Contains(raw, []byte("sig=")) {
		t.Fatal("MCP leaked credential or capability")
	}
	return consistencyJSON(t, mustJSON(out.StructuredContent)), out.IsError
}

func m211AssertError(t *testing.T, value map[string]any, code errcode.Code) {
	t.Helper()
	body, ok := value["error"].(map[string]any)
	if !ok || body["code"] != string(code) || body["request_id"] == "" {
		t.Fatalf("unexpected machine error: %s; want %s", mustJSON(value), code)
	}
}

func m211ReceiptSnapshot(t *testing.T, a *application.App, pushes []m211Push) map[string]any {
	t.Helper()
	out := map[string]any{}
	for _, p := range pushes {
		for _, item := range []struct {
			db  ownership.Database
			key string
		}{{ownership.Runtime, p.State.CreateKey}, {ownership.Ledger, p.State.CommitKey}} {
			store, err := commands.NewStore(map[ownership.Database]string{ownership.Runtime: "storage", ownership.Ledger: "ledger"}[item.db], nil)
			if err != nil {
				t.Fatal(err)
			}
			var op string
			if err = a.Instance.DB(item.db).QueryRowContext(t.Context(), "SELECT operation_id FROM command_receipts WHERE idempotency_key = ?", item.key).Scan(&op); err != nil {
				t.Fatal(err)
			}
			r, err := store.ReceiptByOperation(t.Context(), a.Instance.DB(item.db), ids.ID(op))
			if err != nil || r == nil {
				t.Fatal("missing persisted receipt", err)
			}
			out[p.Adapter+"-"+string(item.db)] = r
		}
		v, err := a.Ledger.Operation(t.Context(), ids.ID(p.State.OperationID))
		if err != nil {
			t.Fatal(err)
		}
		out[p.Adapter+"-operation"] = v
		versions, err := a.Ledger.AssetVersions(t.Context(), ids.ID(p.Result["asset_id"].(string)), 0, 100)
		if err != nil || len(versions) != 1 || versions[0].VersionNumber != 1 {
			t.Fatal("replay created a second version", err, len(versions))
		}
	}
	return out
}

func m211Tree(t *testing.T, root string) map[string]any {
	t.Helper()
	out := map[string]any{}
	err := filepath.WalkDir(root, func(path string, entry os.DirEntry, err error) error {
		if err != nil {
			return err
		}
		if entry.IsDir() {
			return nil
		}
		if !entry.Type().IsRegular() {
			return fmt.Errorf("nonregular archive member")
		}
		b, err := os.ReadFile(path)
		if err != nil {
			return err
		}
		rel, _ := filepath.Rel(root, path)
		out[filepath.ToSlash(rel)] = map[string]any{"size": len(b), "sha256": shaOf(b)}
		return nil
	})
	if err != nil {
		t.Fatal(err)
	}
	return out
}
